⚙️ 算法流程

# 心理数字孪生完整流水线

# 阶段 1：因果图发现（每 N 小时更新一次）
for each user:
  time_series = collect_historical_observations(user)
  causal_graph = PersonalCausalDiscovery().discover(user_id, time_series)
  # 输出：circadian →(lag=2, p=0.01)→ cognitive →(lag=1, p=0.02)→ text_emotion

# 阶段 2：实时状态追踪（每次感知更新）
def on_perception_update(user_id, observation):
  twin = get_digital_twin(user_id)
  twin.update(observation) # 因果感知状态空间模型
  # 内部：predicted = A_causal @ state → adaptive_alpha → new_state

# 阶段 3：轨迹预测 + 干预优化（按需触发）
  trajectory = twin.predict_trajectory(horizon=12) # 预测未来 12 步
  decision = InterventionTimingOptimizer().optimize(twin)
  # 输出：should_intervene=True, optimal_delay=2, chain_to_break="circadian→cognitive"
